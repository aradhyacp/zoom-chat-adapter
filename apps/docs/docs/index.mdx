  intro="Two files: the app, and the route Zoom posts to. Hand the request over untouched — reading the body first breaks signature checks for every event."

<div class="lp-split">
<figure>
<figcaption>lib/bot.ts</figcaption>

```ts
import { createMemoryState } from "@chat-adapter/state-memory";
import { Chat } from "chat";
import { createZoomAdapter } from "zoom-chat-adapter";

export const bot = new Chat({
  userName: "deploybot",
  adapters: { zoom: createZoomAdapter() },
  state: createMemoryState(),
});

bot.onNewMention(async (thread, message) => {
  await thread.post(`Looking into ${message.text}…`);
});
```

</figure>
<figure>
<figcaption>app/api/webhooks/zoom/route.ts</figcaption>

```ts
import { after } from "next/server";
import { bot } from "@/lib/bot";

export async function POST(request: Request) {
  return bot.webhooks.zoom(request, { waitUntil: after });
}
```

Credentials come from the environment, so `createZoomAdapter()` takes no arguments in most apps. A missing one throws at startup and names the variable to set.
